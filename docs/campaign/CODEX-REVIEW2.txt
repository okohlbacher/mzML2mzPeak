   202	            break;
   203	        }

   175	/// Given the document head, return (UTF-8 prolog bytes with `encoding="UTF-8"`, remaining body
   176	/// bytes that followed the prolog). If no prolog is present, the prolog output is empty and the
   177	/// whole head is returned as body.
   178	fn rewrite_prolog(head: &[u8]) -> (Vec<u8>, &[u8]) {
   179	    let text = String::from_utf8_lossy(head);
   180	    let Some(close_rel) = text.find("?>") else {
   181	        // No complete prolog in the head — treat all of it as body (rare; tiny files).
   182	        return (Vec::new(), head);
   183	    };
   184	    let close = close_rel + 2;
   185	    let prolog = &text[..close];

   187	        (Some(enc_at), _) => {

   189	            let after_kw = &prolog[enc_at + "encoding".len()..];
   190	            if let Some(eq) = after_kw.find('=') {

   197	                    format!("{pre}{quote}UTF-8{quote}{tail}")
   198	                } else {

   200	                }
   201	            } else {
   202	                prolog.to_string()
   203	            }
   204	        }
   205	        _ => prolog.to_string(),

   123	    /// Like [`ImagingReader::open`] but with the opt-in checksum escape hatch threaded into the
   124	    /// preflight (CLI `--allow-checksum-mismatch`). When `allow_checksum_mismatch` is true, a
   125	    /// whole-file `.ibd` checksum mismatch warns instead of failing; the UUID linkage is still
   126	    /// enforced. The reader path is otherwise identical (incl. the Latin-1 transcode shim).
   127	    pub fn open_with(
   128	        imzml_path: &Path,
   129	        allow_checksum_mismatch: bool,
   130	    ) -> Result<ImagingReader, ReadError> {
   131	        // (1) Integrity gate BEFORE any reader construction.
   132	        let report = preflight::preflight_with(imzml_path, allow_checksum_mismatch)?;
   133	
   134	        // (2) Open the mzdata reader (preflight already proved the .ibd linkage). If the imzML
   135	        //     declares a non-UTF-8 XML encoding (ISSUE-2: e.g. ISO-8859-1 DESI files), mzdata's

   137	        //     and hand the reader (temp XML, untouched original .ibd) explicitly. The `.ibd`
   138	        //     offsets are ASCII in the XML and the UUID text is preserved, so neither the data
   139	        //     nor mzdata's `.ibd` linkage check is affected.

   142	            transcode::XmlEncoding::Utf8 => (
   143	                ImzMLReader::<File, File>::open_path(imzml_path).map_err(ReadError::Open)?,
   144	                None,
   145	            ),

   152	                (ImzMLReader::<File, File>::new(xml_file, ibd_file), Some(guard))

   154	        };

   157	        let md = &inner.imzml_metadata;

   173	            ibd_checksum: md.ibd_checksum.clone(),
